// The Sims 2 PSP - hand recovered symbol names.
//
// The EBOOT ships with its symbol table stripped, so every name here was
// recovered by analysis rather than read out of the binary.  This file is
// merged over the generated `func_XXXXXXXX` placeholders by
// tools/gen_splat_config.py; the format matches config/eboot.symbol_addrs.txt.
//
// Keep entries in address order and always give the reason, so the next
// person can tell a real name from a guess.

// ---------------------------------------------------------------------------
// File format (Elem, the Maxis engine shared with The Sims 3)
// ---------------------------------------------------------------------------
// Unpacks the packed four-character tag `tag` into a C string and adds it to
// the chunk table of the loader at 0x001276A8, together with the tag's length
// and alignment.  Every static constructor calls this, which is what makes
// `.cplinit` a map of the engine's file formats.
elem_register_chunk_tag = 0x00106D84;

// Calls an allocator and, on failure, raises the C++ bad_alloc message
// "Out of memory -- allocation of %d bytes failed." - the replacement
// `operator new` every `elem_register_chunk_tag` record goes through.
elem_operator_new = 0x001276A8;

// Raises that bad_alloc: takes the message pointer in $a0 and the requested
// size in $a1.
elem_throw_bad_alloc = 0x0012EF70;
